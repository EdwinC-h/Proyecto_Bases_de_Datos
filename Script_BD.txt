create database flota_de_buses;
use flota_de_buses;

create table pasajeros(
	id_pasajero int primary key auto_increment,
    nombre varchar(200) not null,
    apellido varchar(200) not null,
    email varchar (250) unique,
    telefono varchar (50) not null
);

create table rutas(
	id_ruta int primary key auto_increment,
    origen varchar(350) not null,
    destino varchar(350) not null,
    hora_salida datetime default current_timestamp not null,
    hora_llegada datetime not null
);


create table conductores(
	id_conductor int primary key auto_increment,
    nombre varchar(200) not null,
    cedula varchar(20) not null unique,
    telefono varchar(20) not null
);

create table buses(
	id_bus int primary key auto_increment,
    modelo varchar (100) not null,
    placa varchar (20) not null unique,
    año int not null check(año > 2000),
    capacidad int not null check(capacidad between 1 and 80)
);

create table asignaciones(
	id_asignacion int primary key auto_increment,
    id_ruta int,
    id_bus int,
    id_conductor int,
	fecha datetime default current_timestamp,
    foreign key (id_ruta) references rutas(id_ruta),
    foreign key (id_bus) references buses(id_bus),
    foreign key (id_conductor) references conductores(id_conductor)
);

create table boletos(
	id_boleto int primary key auto_increment,
    id_pasajero int,
    id_ruta int,
    id_bus int,
    asiento int check(asiento between 1 and 60) not null,
    precio decimal(5,2) check(precio between 1 and 100.00) not null,
    fecha datetime default current_timestamp,
    foreign key (id_pasajero) references pasajeros(id_pasajero),
    foreign key (id_ruta) references rutas(id_ruta),
    foreign key (id_bus) references buses(id_bus)
);

-- CONSULTAS ENUNCIADO 8
-- Listar todos los pasajeros de apellido 'Guamán'
SELECT * 
FROM pasajeros 
WHERE apellido = 'Guamán';

-- Mostrar todas las rutas que salen desde Quito
select origen, destino, hora_salida, hora_llegada
from rutas
where origen = "Quito";

-- pasajeros con sus boletos y rutas
select p.nombre, p.apellido, b.asiento, r.origen, r.destino
from pasajeros p
join boletos b on p.id_pasajero = b.id_pasajero
join rutas r on b.id_ruta = r.id_ruta;

-- conductores asignados a buses y rutas
select c.nombre, bu.modelo, r.origen, r.destino, a.fecha
from conductores c
join asignaciones a on c.id_conductor = a.id_conductor
join buses bu on a.id_bus = bu.id_bus
join rutas r on a.id_ruta = r.id_ruta;

-- boletos vendidos por bus
select bu.placa, count(b.id_boleto) as boletos_vendidos
from buses bu
join boletos b on bu.id_bus = b.id_bus
group by bu.placa;

-- reporte de ocupación por ruta
select r.origen, r.destino, count(b.id_boleto) as pasajeros
from rutas r
join boletos b on r.id_ruta = b.id_ruta
group by r.origen, r.destino;

-- pasajeros y el conductor que los transportó
select p.nombre as pasajero, c.nombre as conductor, r.origen, r.destino
from pasajeros p
join boletos b on p.id_pasajero = b.id_pasajero
join rutas r on b.id_ruta = r.id_ruta
join asignaciones a on r.id_ruta = a.id_ruta and b.id_bus = a.id_bus
join conductores c on a.id_conductor = c.id_conductor;

-- ingresos por bus
select bu.placa, sum(b.precio) as ingresos
from buses bu
join boletos b on bu.id_bus = b.id_bus
group by bu.placa;

-- Ingresos totales por dia
select date(fecha) as dia, sum(precio) as ingresos
from boletos
group by date(fecha);

-- Promedio de precio de boletos por ruta 
select r.origen, r.destino, avg(b.precio) as precio_promedio
from rutas 
join boletos b on r.id_ruta = b.id_ruta
group by r.origen, r.destino;

-- Mostrar correos en minúsculas
select lower(email) as correo_normalizado
from pasajeros;

-- Concatenar nombre completo de pasajeros
select concat(nombre, ' ', apellido) as nombre_completo
from pasajeros;

-- Subconsulta: pasajeros que han comprado boletos más caros que el promedio
select nombre, apellido
from pasajeros
where id_pasajero in(select id_pasajero from boletos where precio > (select avg(precio) from boletos));

-- Vista: reporte de ocupación por ruta
create view reporte_ocupacion as 
select r.origen, r.destino, COUNT(b.id_boleto) as pasajeros
from rutas r 
join boletos b on r.id_ruta = b.id_ruta
group by r.origen, r.destino;
